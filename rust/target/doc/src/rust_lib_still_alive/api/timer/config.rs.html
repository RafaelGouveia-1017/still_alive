4/// Configuration for a timer.
5#[derive(Debug, Clone, Serialize, Deserialize)]
6#[frb(opaque)]
7pub struct TimerConfig {
8    /// Human-readable name of the timer.
9    pub name: String,
10
11    /// Timer duration in seconds.
12    pub duration_secs: i64,
13
14    /// Optional grace period in seconds after the timer expires.
15    pub grace_period_secs: Option<i64>,
16
17    /// Whether the timer requires password verification for protected operations.
18    pub password_protected: bool,
19
20    /// Stored password hash used to verify protected operations.
21    pub password_hash: Option<String>,
22
23    /// Whether the timer should obtain the device's location when an emergency
24    /// is triggered.
25    ///
26    /// When this is `true` and [routeSharingEnabled] is `false`, the application
27    /// obtains the device's current location at emergency time and shares that
28    /// single location.
29    ///
30    /// When [routeSharingEnabled] is `true`, location collection is implicitly
31    /// enabled and the application records a GPS route throughout the active
32    /// timer run instead of obtaining only a single location at emergency time.
33    pub location_sharing_enabled: bool,
34
35    /// Whether the timer should record and share a complete GPS route.
36    ///
37    /// When this is `true`, location collection is implicitly enabled. The
38    /// [locationCollectionIntervalSecs] value determines the minimum interval
39    /// between persisted GPS route points.
40    ///
41    /// When this is `false` but [locationSharingEnabled] is `true`, no continuous
42    /// route is recorded. Instead, the application's current location is obtained
43    /// when an emergency is triggered and that single location is shared.
44    ///
45    /// When both [locationSharingEnabled] and [routeSharingEnabled] are `false`,
46    /// no location is collected or shared.
47    pub route_sharing_enabled: bool,
48
49    /// The minimum number of seconds between persisted GPS route points.
50    ///
51    /// This value is only used when [routeSharingEnabled] is `true`.
52    ///
53    /// When route sharing is disabled, this value has no effect because the
54    /// application does not continuously record the device's location.
55    pub location_collection_interval_secs: Option<i64>,
56
57    /// Whether audio recording is enabled for this timer.
58    pub audio_recording_enabled: bool,
59
60    /// Contacts associated with the timer.
61    pub contacts: Vec<Contact>,
62
63    /// Custom SMS messages associated with the timer.
64    pub custom_sms: Vec<String>,
65
66    /// Custom email messages associated with the timer.
67    pub custom_email: Vec<String>,
68
69    /// External service integrations configured for the timer.
70    pub integrations: TimerIntegrations,
71
72    /// Custom text to send to the destinations.
73    pub message: Option<String>,
74
75    /// Timestamp or serialized date representing when the timer configuration was created.
76    pub created_at: String,
77
78    /// Timestamp or serialized date representing when the timer configuration was last updated.
79    pub updated_at: String,
80}
