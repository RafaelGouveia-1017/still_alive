10/// Export a complete application backup as a ZIP archive in memory.
11///
12/// The generated backup contains:
13///
14/// - the SQLite database
15/// - a metadata file
16///
17/// The ZIP archive is returned as raw bytes (`Vec<u8>`), making it
18/// platform-independent.
19///
20/// # Export procedure
21///
22/// 1. Perform WAL checkpoint to ensure DB consistency.
23/// 2. Read database file into memory.
24/// 3. Generate metadata JSON in memory.
25/// 4. Create ZIP archive in memory.
26/// 5. Return ZIP as `Vec<u8>`.
27///
28/// The database connection remains open throughout the export.
29///
30/// # Arguments
31///
32/// * `app_version` - Application version.
33///
34/// # Returns
35///
36/// A `Vec<u8>` containing the full ZIP archive.
37///
38/// # Errors
39///
40/// Returns an error if:
41///
42/// - WAL checkpoint fails
43/// - database file cannot be read
44/// - metadata cannot be generated
45/// - ZIP creation fails
46///
47/// # Notes
48///
49/// - No files are written to storage.
50/// - Entire ZIP is held in memory.
51pub fn export_backup(app_version: &str) -> Result<Vec<u8>> {
52    let db = db();
53
54    let database = db
55        .as_ref()
56        .ok_or_else(|| anyhow!("database not initialized"))?;
57
58    database.checkpoint()?;
59
60    let db_bytes = fs::read(database_path_str())?;
61
62    let metadata_json = create_backup_metadata_string(app_version)?;
63
64    let mut buffer = Vec::new();
65    let cursor = Cursor::new(&mut buffer);
66    let mut zip = ZipWriter::new(cursor);
67
68    let options = SimpleFileOptions::default().compression_method(zip::CompressionMethod::Deflated);
69
70    zip.start_file(get_database_name(), options)?;
71    zip.write_all(&db_bytes)?;
72
73    zip.start_file("metadata.json", options)?;
74    zip.write_all(metadata_json.as_bytes())?;
75
76    let log_path = database_path_str()
77        .split(&get_database_name())
78        .next()
79        .unwrap();
80    let log_bytes = fs::read(log_path.to_owned() + "app.log")?;
81
82    zip.start_file("app.log", options)?;
83    zip.write_all(&log_bytes)?;
84
85    let cursor = zip.finish()?;
86    let zip_buffer = cursor.into_inner().to_vec();
87
88    Ok(zip_buffer)
89}

91/// Import an application backup from a ZIP archive.
92///
93/// # Import procedure:
94///
95/// 1. Extract ZIP archive into memory.
96/// 2. Extract `metadata.json` and validate it.
97/// 3. Validate backup format and compatibility.
98/// 4. Extract database file into memory.
99/// 5. Close current database connection.
100/// 6. Overwrite existing database file.
101/// 7. Reopen database.
102///
103/// # Arguments
104///
105/// - `zip_path` - Path to the backup ZIP archive.
106/// - `app_version` - Current application version.
107///
108/// # Errors
109///
110/// Returns an error if:
111///
112/// - the ZIP archive is invalid
113/// - required files are missing
114/// - metadata validation fails
115/// - the database cannot be replaced
116pub fn import_backup(zip_path: &str, app_version: &str) -> Result<()> {
117    let file = File::open(zip_path)?;
118
119    let mut archive = ZipArchive::new(file)?;
120
121    {
122        let mut metadata_file = archive
123            .by_name("metadata.json")
124            .map_err(|_| anyhow!("metadata.json not found"))?;
125
126        let mut metadata_str = String::new();
127        metadata_file.read_to_string(&mut metadata_str)?;
128
129        let metadata = validate_backup_metadata(&metadata_str)?;
130
131        if metadata.format_version != 1 {
132            return Err(anyhow!("unsupported backup format version"));
133        }
134
135        if metadata.app_version != app_version {
136            return Err(anyhow!("backup created with different app version"));
137        }
138    };
139    {
140        let mut db_file = archive
141            .by_name(&get_database_name())
142            .map_err(|_| anyhow!("database file not found"))?;
143
144        let mut db_bytes = Vec::new();
145        db_file.read_to_end(&mut db_bytes)?;
146
147        close_database()?;
148
149        fs::write(database_path_str(), &db_bytes)?;
150
151        open_database()?;
152    };
153
154    Ok(())
155}
