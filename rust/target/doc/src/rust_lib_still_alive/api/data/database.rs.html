8/// Thin SQLite wrapper built on top of rusqlite.
9///
10/// # Features
11///
12/// - SQLite WAL mode enabled by default.
13/// - Automatic database creation.
14/// - Single-row and multi-row query helpers.
15/// - Explicit WAL checkpoint support.
16///
17/// # Persistence
18///
19/// SQLite automatically persists committed changes to disk.
20///
21/// Since WAL mode is enabled, recent writes may temporarily reside in
22/// the WAL file before being merged back into the main database file.
23/// SQLite performs checkpoints automatically as needed.
24///
25/// Use [`Database::checkpoint`] if you need to force a checkpoint.
26#[frb(ignore)]
27#[derive(Debug)]
28pub struct Database {
29    conn: Connection,
30}

32/// Tracks current database version.
33#[frb(ignore)]
34const DB_VERSION: i32 = 1;

36impl Database {
37    /// Opens (or creates) the SQLite database located at `path`.
38    ///
39    /// This function is the main entry point for interacting with the database.
40    /// It performs the following steps:
41    ///
42    /// 1. Opens or creates the database file.
43    /// 2. Configures SQLite PRAGMAs (WAL mode, foreign keys, etc.).
44    /// 3. Applies any pending schema and data migrations.
45    ///
46    /// After this function returns successfully, the database is guaranteed to
47    /// match the latest schema version (`DB_VERSION`), regardless of which
48    /// application version originally created it.
49    ///
50    /// # Errors
51    ///
52    /// Returns an error if:
53    ///
54    /// - The database cannot be opened.
55    /// - SQLite configuration fails.
56    /// - Any migration fails.
57    pub fn open(path: &str) -> Result<Self> {
58        let conn = Connection::open(path)
59            .inspect_err(|e| panic!("\n\n\nDB connection error: {e}\npath: {}\n\n\n", path))
60            .unwrap();
61
62        conn.execute_batch(
63            r#"
64            PRAGMA journal_mode = WAL;
65            PRAGMA synchronous = NORMAL;
66            PRAGMA foreign_keys = ON;
67            "#,
68        )?;
69
70        let mut version = conn.query_row("PRAGMA user_version;", [], |row| row.get(0))?;
71        while version < DB_VERSION {
72            println!("\nMigrating database: {} -> {}\n", version, version + 1);
73
74            match version {
75                0 => conn.execute_batch(include_str!("../../schema/base.sql"))?,
76                //1 => conn.execute_batch(include_str!("../../schema/migration_v1.0.32.sql"))?,
77                _ => return Err(anyhow!("Unknown database version {}", version)),
78            }
79
80            version += 1;
81            conn.execute(&format!("PRAGMA user_version = {}", version), [])?;
82        }
83
84        Ok(Self { conn })
85    }

87    /// Close the SQLite connection.
88    ///
89    /// This cleanly closes the underlying SQLite connection.
90    ///
91    /// After this method returns successfully:
92    /// - all pending transactions have been committed or rolled back
93    /// - the connection becomes unusable
94    ///
95    /// This method is primarily intended for administrative operations
96    /// such as importing a database backup, where the database file must
97    /// be replaced on disk.
98    ///
99    /// # Errors
100    ///
101    /// Returns an error if SQLite cannot close the connection.
102    /// This can occur if prepared statements are still active.
103    ///
104    /// # Example
105    ///
106    /// ```rust
107    /// let db = Database::open("app.db")?;
108    /// db.close()?;
109    /// ```
110    pub fn close(self) -> Result<()> {
111        self.conn.close().map_err(|(_, err)| anyhow!(err))
112    }

114    /// Export the current SQLite database.
115    ///
116    /// Before copying the database file, this method performs a
117    /// `wal_checkpoint(TRUNCATE)` to flush all committed changes from the
118    /// WAL file into the main database.
119    ///
120    /// The resulting copy is therefore a complete and consistent snapshot.
121    ///
122    /// The database connection remains open throughout the operation.
123    ///
124    /// # Arguments
125    ///
126    /// - `destination` - Destination path of the exported database.
127    ///
128    /// # Errors
129    ///
130    /// Returns an error if:
131    ///
132    /// - the WAL checkpoint fails
133    /// - the database file cannot be copied
134    ///
135    /// # Notes
136    ///
137    /// This function only exports the SQLite database file.
138    ///
139    /// Additional backup files (such as `metadata.json`) are created
140    /// by higher-level backup functions.
141    ///
142    /// # Example
143    ///
144    /// ```rust
145    /// db.export("/tmp/backup.db")?;
146    /// ```
147    pub fn export(&self, destination: &str) -> Result<()> {
148        self.checkpoint()?;
149
150        std::fs::copy(crate::api::data::db::database_path_str(), destination)?;
151
152        Ok(())
153    }

155    /// Delete an existing SQLite database.
156    ///
157    /// This removes:
158    /// - main database file
159    /// - WAL file
160    /// - SHM file
161    ///
162    /// The application should not be actively using the database
163    /// when this function is called.
164    ///
165    /// The application MUST restart after this function is called.
166    pub fn purge(path: &str) {
167        let _ = fs::remove_file(path);
168
169        let wal_path = PathBuf::from(format!("{}-wal", path));
170        let shm_path = PathBuf::from(format!("{}-shm", path));
171
172        let _ = fs::remove_file(wal_path);
173        let _ = fs::remove_file(shm_path);
174    }

203    /// Query a single row.
204    ///
205    /// Returns `None` if no matching row exists.
206    ///
207    /// Example:
208    ///
209    /// ```rust
210    /// let name = db.query_one(
211    ///     "SELECT name FROM users WHERE id = ?",
212    ///     [1],
213    ///     |row| row.get::<_, String>(0),
214    /// )?;
215    /// ```
216    pub fn query_one<T, P, F>(&self, sql: &str, params: P, mapper: F) -> Result<Option<T>>
217    where
218        P: Params,
219        F: FnOnce(&Row<'_>) -> rusqlite::Result<T>,
220    {
221        Ok(self.conn.query_row(sql, params, mapper).optional()?)
222    }

224    /// Queries multiple rows and returns the result as JSON.
225    ///
226    /// # Arguments
227    ///
228    /// * `sql` - A SQL query string. Typically a `SELECT` statement.  
229    /// * `params` - Dynnamic parameters for SQL query.  
230    ///
231    /// # Returns
232    ///
233    /// A JSON string representing a list of rows.  
234    /// Each row is a JSON object where:
235    /// - keys = column names
236    /// - values = column values converted to JSON types:
237    ///   - INTEGER → number
238    ///   - REAL → floating point number
239    ///   - TEXT → string
240    ///   - NULL → null
241    ///   - BLOB → base64 string (if enabled) or null
242    ///
243    /// # Example
244    ///
245    /// ```rust
246    /// let users = db.query_many(
247    ///     "SELECT id, name FROM users WHERE name LIKE %?%",
248    ///     [mark]
249    /// )?;
250    /// ```
251    ///
252    /// Returns:
253    ///
254    /// ```json
255    /// [
256    ///   { "id": 1, "name": "Alice" },
257    ///   { "id": 2, "name": "Bob" }
258    /// ]
259    /// ```
260    ///
261    /// Usage in Dart:
262    ///
263    /// ```dart
264    /// import 'dart:convert';
265    /// (...)
266    /// final jsonStr = await query_many(...);
267    /// final List data = jsonDecode(jsonStr);
268    /// String name = data[0]['name'];
269    /// ```
270    ///
271    /// # Notes
272    ///
273    /// - Column names are extracted dynamically at runtime.
274    /// - This function supports queries across multiple tables (JOINs included).
275    /// - Results are not typed; all output is serialized into JSON.
276    /// - BLOB values may be base64-encoded depending on implementation.
277    /// - Intended for flexible query execution, not strict type safety.
278    ///
279    /// # Safety
280    ///
281    /// This function executes raw SQL provided by the caller.  
282    /// Do not expose it directly to untrusted input without validation or sanitization.
283    /// (i.e: SQL injection)
284    ///
285    /// # Performance
286    ///
287    /// - Suitable for small to medium result sets.
288    /// - Large datasets may require pagination (`LIMIT/OFFSET`) or streaming.
289    ///
290    /// # Panics
291    ///
292    /// This function does not panic. All errors are returned as `Err(String)`.
293    pub fn query_many<P>(&self, sql: &str, params: P) -> Result<String, String>
294    where
295        P: Params,
296    {
297        let mut stmt = self.conn.prepare(sql).map_err(|e| e.to_string()).unwrap();
298
299        let column_names: Vec<String> = (0..stmt.column_count())
300            .map(|i| stmt.column_name(i).unwrap_or("").to_string())
301            .collect();
302
303        let rows_iter = stmt
304            .query_map(params, |row| {
305                let mut obj = serde_json::Map::new();
306
307                for (i, name) in column_names.iter().enumerate() {
308                    let value: Value = row.get(i)?;
309
310                    let json_value = match value {
311                        Value::Null => serde_json::Value::Null,
312                        Value::Integer(v) => json!(v),
313                        Value::Real(v) => json!(v),
314                        Value::Text(v) => json!(v),
315                        Value::Blob(_) => serde_json::Value::Null, // Value::Blob(b) => serde_json::Value::String(base64::encode(b)), //if it's an image or a file (must install base64 package)
316                    };
317
318                    obj.insert(name.clone(), json_value);
319                }
320
321                Ok(serde_json::Value::Object(obj))
322            })
323            .map_err(|e| e.to_string())?;
324
325        let mut rows = Vec::new();
326
327        for r in rows_iter {
328            rows.push(r.map_err(|e| e.to_string())?);
329        }
330
331        serde_json::to_string(&rows).map_err(|e| e.to_string())
332    }

334    /// Force a WAL checkpoint.
335    ///
336    /// Normally SQLite performs checkpoints automatically.
337    ///
338    /// Useful when:
339    ///
340    /// - Preparing to copy the database file.
341    /// - Reclaiming WAL disk space.
342    /// - Administrative maintenance.
343    pub fn checkpoint(&self) -> Result<()> {
344        self.conn
345            .execute_batch("PRAGMA wal_checkpoint(TRUNCATE);")?;

375        let db = Database::open(":memory:")?;
376
377        db.execute_batch("CREATE TABLE users (id INTEGER PRIMARY KEY, name TEXT NOT NULL);")?;
378
379        db.execute("INSERT INTO users (name) VALUES (?)", ["Alice"])?;
380        db.execute("INSERT INTO users (name) VALUES (?)", ["Bob"])?;
381
382        let alice_name = db.query_one("SELECT name FROM users WHERE id = ?", [1], |row| {

384        })?;

387        let json_all = db

396        assert_eq!(parsed_all[0]["name"], "Alice");
