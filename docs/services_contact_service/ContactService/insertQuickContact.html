Adds contactID to the list of quick contacts.

If contactID is already present, the list is left unchanged. Otherwise, the contact ID is appended to the ids array stored in the quick row. The count field is then updated to reflect the number of unique stored IDs.

Implementation

static Future<void> insertQuickContact(String contactID) async {
  await executeBatchSql(
    sql:
        """
        UPDATE contacts
        SET value = json_insert(value, '\$.ids[#]', '$contactID')
        WHERE key = 'quick'
          AND NOT EXISTS (
              SELECT 1
              FROM json_each(contacts.value, '\$.ids')
              WHERE json_each.value = '$contactID'
          );
        UPDATE contacts
        SET value = json_set(
            value,
            '\$.count',
            json_array_length(json_extract(value, '\$.ids'))
        )
        WHERE key = 'quick';
        """,
  );
}
